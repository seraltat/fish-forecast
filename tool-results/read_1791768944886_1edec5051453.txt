    17→// ----------------- Piecewise-linear interpolation -----------------
    18→
    19→// points sorted ascending by [0]; linear interp between; clamp ends.
    20→export function piecewiseLinear(points: Array<[number, number]>, x: number): number {
    21→  if (points.length === 0) return 0;
    22→  if (x <= points[0][0]) return points[0][1];
    23→  const last = points[points.length - 1];
    24→  if (x >= last[0]) return last[1];
    25→  for (let i = 0; i < points.length - 1; i++) {
    26→    const [x0, y0] = points[i];
    27→    const [x1, y1] = points[i + 1];
    28→    if (x >= x0 && x <= x1) {
    29→      const t = (x - x0) / (x1 - x0);
    30→      return Math.round(y0 + t * (y1 - y0));
    31→    }
    32→  }
    33→  return last[1];
    34→}

    42→export function moonPhase(date: Date): { age: number; phase: number; label: string } {
    43→  const ms = date.getTime() - LUNAR_EPOCH_MS;
    44→  const day = ms / (1000 * 60 * 60 * 24);
    45→  const age = ((day % SYNODIC_MONTH) + SYNODIC_MONTH) % SYNODIC_MONTH;
    46→  const phase = age / SYNODIC_MONTH; // 0..1
    47→  return { age, phase, label: moonPhaseLabel(phase) };
    48→}

   106→export type PressureTrend = "drop_sharp" | "drop" | "stable" | "rise" | "rise_sharp";

   108→// Thresholds based on user's research (image 1):
   109→//   Резкий скачок = падение ≥ 5–8 mmHg/сутки → ~0.6–1 mmHg/3ч
   110→//   Плавное падение = 2–3 mmHg/сутки → ~0.25–0.4 mmHg/3ч
   111→// Thresholds (per 3-hour window):
   112→//   < -1 mmHg/3ч           → Резко падает
   113→//   -1 .. -0.3 mmHg/3ч     → Падает (плавное)
   114→//   -0.3 .. +0.3 mmHg/3ч   → Стабильно
   115→//   +0.3 .. +1 mmHg/3ч     → Растёт (плавное)
   116→//   > +1 mmHg/3ч           → Резко растёт
   117→export function pressureTrend(deltaMmHg: number): { trend: PressureTrend; label: string } {
   118→  if (deltaMmHg < -1) return { trend: "drop_sharp", label: "Резко падает" };
   119→  if (deltaMmHg < -0.3) return { trend: "drop", label: "Падает" };
   120→  if (deltaMmHg <= 0.3) return { trend: "stable", label: "Стабильно" };
   121→  if (deltaMmHg <= 1) return { trend: "rise", label: "Растёт" };
   122→  return { trend: "rise_sharp", label: "Резко растёт" };
   123→}

   241→// Bell-shaped sub-score kept for backward compat (still used in some tests).
   242→export function bellScore(value: number, peak: number, lo: number, hi: number): number {
   243→  if (value < lo || value > hi) {
   244→    return Math.max(0, 100 - Math.abs(value - peak) * 8);
   245→  }
   246→  const half = Math.max((hi - lo) / 2, 0.001);
   247→  const d = Math.abs(value - peak) / half;
   248→  return Math.round(100 - 25 * d * d);
   249→}

   283→// Wind direction sub-score — piecewise by 8-point compass.
   284→// По правилу пользователя: «южный, западный клев нормальный; северный, восточный — не очень».
   285→// Используем 8-румбовый компас с явной оценкой для каждого направления:
   286→//   С  (N,   0°)   → 25  — холодный, не очень
   287→//   СВ (NE, 45°)  → 30  — холодный с востоком, не очень
   288→//   В  (E,  90°)  → 35  — восточный, не очень
   289→//   ЮВ (SE, 135°) → 75  — тёплый с востоком, хорошо
   290→//   Ю  (S, 180°)  → 90  — тёплый, хороший
   291→//   ЮЗ (SW, 225°) → 100 — лучший: тёплый, нагоняет корм в берег
   292→//   З  (W, 270°)  → 90  — хороший
   293→//   СЗ (NW, 315°) → 65  — западный с холодным севером, средне
   294→const WIND_DIR_SCORES_8PT: Array<{ short: string; score: number; deg: number }> = [
   295→  { short: "С", score: 25, deg: 0 },
   296→  { short: "СВ", score: 30, deg: 45 },
   297→  { short: "В", score: 35, deg: 90 },
   298→  { short: "ЮВ", score: 75, deg: 135 },
   299→  { short: "Ю", score: 90, deg: 180 },
   300→  { short: "ЮЗ", score: 100, deg: 225 },
   301→  { short: "З", score: 90, deg: 270 },
   302→  { short: "СЗ", score: 65, deg: 315 },
   303→];
   304→
   305→export function windDirScore(deg: number | null | undefined, _bestDirs?: number[]): number {
   306→  if (deg == null || Number.isNaN(deg)) return 50;
   307→  const norm = ((deg % 360) + 360) % 360;
   308→  // Find closest 8-point compass direction
   309→  let closest = WIND_DIR_SCORES_8PT[0];
   310→  let bestDiff = 360;
   311→  for (const dir of WIND_DIR_SCORES_8PT) {
   312→    const diff = Math.min(Math.abs(norm - dir.deg), 360 - Math.abs(norm - dir.deg));
   313→    if (diff < bestDiff) {
   314→      bestDiff = diff;
   315→      closest = dir;
   316→    }
   317→  }
   318→  // If exactly on the compass point, return its score.
   319→  // If between two points, interpolate linearly between their scores.
   320→  if (bestDiff === 0) return closest.score;
   321→  // Find the next compass point clockwise
   322→  const idx = WIND_DIR_SCORES_8PT.indexOf(closest);
   323→  const next = WIND_DIR_SCORES_8PT[(idx + 1) % 8];
   324→  const nextDeg = next.deg < closest.deg ? next.deg + 360 : next.deg;
   325→  const segmentSize = nextDeg - closest.deg;
   326→  const t = bestDiff / segmentSize;
   327→  return Math.round(closest.score + t * (next.score - closest.score));
   328→}

   374→// Принимает ЛУННЫЙ ДЕНЬ (1..30) — по русскому рыбацкому календарю,
   375→// где день начинается в момент восхода луны, а не в астрономическое новолуние.
   376→// Это привязывает день к координатам (восход луны зависит от широты/долготы).
   377→export function moonPhaseScore(lunarDay: number): number {
   378→  const day = Math.max(1, Math.min(30, Math.floor(lunarDay)));
   379→  if (day === 1 || day === 2) return 25;       // плохой (новолуние)
   380→  if (day === 3) return 70;                    // хороший
   381→  if (day >= 4 && day <= 8) return 90;          // отличный
   382→  if (day === 9 || day === 10) return 70;       // хороший
   383→  if (day === 11 || day === 12) return 45;      // слабый
   384→  if (day >= 13 && day <= 16) return 25;        // плохой (полнолуние)
   385→  if (day === 17) return 45;                    // слабый
   386→  if (day === 18 || day === 19) return 90;      // отличный
   387→  if (day >= 20 && day <= 22) return 70;        // хороший
   388→  if (day >= 23 && day <= 26) return 45;        // слабый
   389→  if (day >= 27) return 25;                    // плохой (близко к новолунию)
   390→  return 50; // fallback

   393→// Compute Russian-fishing-tradition lunar day (1..30) for a given moment
   394→// based on:
   395→//   1. Astronomical new moon (computed from lunar epoch 2000-01-06 18:14 UTC)
   396→//   2. List of moonrise times at the user's location (from Open-Meteo Astronomy API)
   397→//
   398→// Algorithm:
   399→//   - Find the most recent new moon (date) before `now`
   400→//   - Count moonrises that happened AFTER new moon AND BEFORE `now`
   401→//   - lunarDay = count + 1
   402→//   (Day 1 = short day from new moon moment to first moonrise)
   403→//   (Day N = from (N-1)th to Nth moonrise after new moon)
   404→//
   405→// Parameters:
   406→//   now — Date constructed from local time string treated as UTC (so getUTCHours() gives local hour)
   407→//   moonrises — array of Date objects (any order). If empty, heuristic is used.
   408→//   phase — moon phase (0..1) computed once per day at noon UTC (consistent across all hours of the day).
   409→//            This is critical: recomputing phase from `now` would give a wrong value because
   410→//            `now` represents local time (treated as UTC), not actual UTC moment.
   411→export function computeUserLunarDay(now: Date, moonrises: Date[], phase: number): number {
   412→  if (moonrises.length === 0) {
   413→    // Heuristic fallback: estimate moonrise from phase.
   414→    // At new moon (phase 0): moonrise ~ sunrise (~06:00 local).
   415→    // At full moon (phase 0.5): moonrise ~ sunset (~18:00 local).
   416→    // Linear interp: moonrise_hour = 6 + phase * 24 (mod 24).
   417→    const age = phase * SYNODIC_MONTH;
   418→    const astroDay = Math.floor(age) + 1;
   419→    const moonriseHour = (6 + phase * 24) % 24; // local hour at spot
   420→    // now.getUTCHours() gives the local hour at the spot because we constructed
   421→    // the Date from Open-Meteo local time strings treated as UTC.
   422→    const localHour = now.getUTCHours() + now.getUTCMinutes() / 60;
   423→    // Account for moonrise that may be early morning (e.g. 02:00) and now is 22:00
   424→    // — that means we're past moonrise, so add 1.
   425→    if (localHour >= moonriseHour) {
   426→      return astroDay + 1;
   427→    }
   428→    return astroDay;
   429→  }
   430→
   431→  // Find most recent new moon before `now`.
   432→  const newMoonDate = mostRecentNewMoon(now);
   433→  // Count moonrises strictly AFTER new moon AND BEFORE (or at) now.
   434→  const count = moonrises.filter(
   435→    (mr) => mr.getTime() > newMoonDate.getTime() && mr.getTime() <= now.getTime()
   436→  ).length;
   437→
   438→  return count + 1;
   439→}
   440→
   441→// Find the most recent new moon before `now` using the lunar epoch + synodic month.
   442→// Iterates backwards from `now` to find the largest t such that
   443→// t = LUNAR_EPOCH_MS + k * SYNODIC_MONTH * ms_per_day, t <= now.
   444→export function mostRecentNewMoon(now: Date): Date {
   445→  const msPerDay = 1000 * 60 * 60 * 24;
   446→  const msPerCycle = SYNODIC_MONTH * msPerDay;
   447→  const elapsedMs = now.getTime() - LUNAR_EPOCH_MS;
   448→  const cycles = Math.floor(elapsedMs / msPerCycle);
   449→  return new Date(LUNAR_EPOCH_MS + cycles * msPerCycle);
   450→}

   452→// Wind speed sub-score: ideal [lo,hi], >badMps → low
   453→export function windSpeedScore(speed: number, lo: number, hi: number, badMps: number): number {
   454→  if (speed >= lo && speed <= hi) return 95;
   455→  if (speed > badMps) return 10;
   456→  if (speed < lo) {
   457→    const d = lo - speed;
   458→    return Math.max(20, Math.round(95 - d * 10));
   459→  }
   460→  const d = speed - hi;
   461→  return Math.max(10, Math.round(95 - d * 15));
   462→}
   463→
   464→// Cloud cover sub-score (0..100%)
   465→export function cloudCoverScore(cover: number, preference: "overcast" | "sunny" | "neutral"): number {
   466→  if (preference === "neutral") return 70;
   467→  if (preference === "overcast") {
   468→    return Math.round((cover / 100) * 90 + 10);
   469→  }
   470→  return Math.round(((100 - cover) / 100) * 90 + 10);
   471→}

   535→// Per-species scoring function
   536→export function scoreForSpecies(species: Species, weights: FishWeights, input: HourInput): number {
   537→  const subscores: Record<string, number> = {};
   538→
   539→  // Water temperature — piecewise curve per species.
   540→  if (weights.waterTemp) {
   541→    subscores.waterTemp = waterTempScore(species, input.waterTemp);
   542→  }
   543→
   544→  // Pressure trend — table-driven per species.
   545→  if (weights.pressureTrend) {
   546→    subscores.pressureTrend = pressureTrendScore(input.pressureTrend, species);
   547→  }
   548→
   549→  // Pressure value — same formula for all species (broad band, minor factor).
   550→  if (weights.pressureValue) {
   551→    subscores.pressureValue = pressureValueScore(input.pressureMmHg);
   552→  }
   553→
   554→  // Wind speed
   555→  if (weights.windSpeed) {
   556→    if (species === "сом") {
   557→      subscores.windSpeed = windSpeedScore(input.windSpeed, 1, 3, 5);
   558→    } else if (species === "щука") {
   559→      subscores.windSpeed = windSpeedScore(input.windSpeed, 2, 5, 8);
   560→    } else if (species === "карась") {
   561→      subscores.windSpeed = windSpeedScore(input.windSpeed, 1, 4, 7);
   562→    } else if (species === "сазан") {
   563→      subscores.windSpeed = windSpeedScore(input.windSpeed, 1, 5, 8);
   564→    }
   565→  }
   566→
   567→  // Wind direction
   568→  if (weights.windDirection) {
   569→    subscores.windDirection = windDirScore(input.windDir, [180, 225, 270]);
   570→  }
   571→
   572→  // Time of day
   573→  if (weights.timeOfDay) {
   574→    if (species === "сом") {
   575→      if (input.isNight) {
   576→        subscores.timeOfDay = 95;
   577→      } else {
   578→        subscores.timeOfDay = Math.max(20, timeOfDayScore(input.minutesFromSunEvent));
   579→      }
   580→    } else {
   581→      subscores.timeOfDay = timeOfDayScore(input.minutesFromSunEvent);
   582→    }
   583→  }
   584→
   585→  // Moon phase — одинаковая таблица 28-дневного цикла для всех видов рыб.
   586→  // Используем lunarDay (по восходам луны на координатах места) вместо
   587→  // глобальной фазы — это даёт корректный лунный день для русского календаря.
   588→  if (weights.moonPhase) {
   589→    subscores.moonPhase = moonPhaseScore(input.lunarDay);
   590→  }
   591→
   592→  // Cloud cover (only щука uses this)
   593→  if (weights.cloudCover) {
   594→    subscores.cloudCover = cloudCoverScore(input.cloudCover, "overcast");
   595→  }
   596→
   597→  // Weighted average
   598→  let total = 0;
   599→  let weightSum = 0;
   600→  for (const [key, w] of Object.entries(weights)) {
   601→    const wnum = w as number;
   602→    if (wnum && subscores[key] != null) {
   603→      total += wnum * subscores[key];
   604→      weightSum += wnum;
   605→    }
   606→  }
   607→  if (weightSum === 0) return 0;
   608→  let finalScore = Math.round(total / weightSum);
   609→
   610→  // Water-temperature gate — критически важная защита от абсурдных оценок.
   611→  // Если вода слишком холодная (или слишком тёплая) — рыба физически не активна,
   612→  // даже при идеальном давлении/ветре/времени суток. Множитель 0..1:
   613→  //   sub-score 50 (= порог активности) → gate 1.0 (без штрафа)
   614→  //   sub-score 25 (слабая активность)  → gate 0.5
   615→  //   sub-score 5  (рыба спит)         → gate 0.1 (сильный штраф)
   616→  // Это делает температуру воды «вратами» для клёва, а не просто одним из факторов.
   617→  if (subscores.waterTemp != null) {
   618→    const tempGate = Math.min(1, subscores.waterTemp / 50);
   619→    finalScore = Math.round(finalScore * tempGate);
   620→  }
   621→
   622→  return finalScore;
   623→}

   625→// Top contributing factors (for "Почему так?" tooltip).
   626→// Returns top 2 factors by weighted contribution * (subscore - 50).
   627→export function topFactors(species: Species, weights: FishWeights, input: HourInput): string[] {
   628→  const results: { factor: string; contribution: number }[] = [];
   629→
   630→  const w = weights;
   631→  if (w.waterTemp) {
   632→    const s = waterTempScore(species, input.waterTemp);
   633→    results.push({ factor: "Температура воды", contribution: (w.waterTemp as number) * (s - 50) });
   634→  }
   635→  if (w.pressureTrend) {
   636→    const s = pressureTrendScore(input.pressureTrend, species);
   637→    results.push({ factor: "Тенденция давления", contribution: (w.pressureTrend as number) * (s - 50) });
   638→  }
   639→  if (w.pressureValue) {
   640→    const s = pressureValueScore(input.pressureMmHg);
   641→    results.push({ factor: "Значение давления", contribution: (w.pressureValue as number) * (s - 50) });
   642→  }
   643→  if (w.windSpeed) {
   644→    let s: number;
   645→    if (species === "сом") s = windSpeedScore(input.windSpeed, 1, 3, 5);
   646→    else if (species === "щука") s = windSpeedScore(input.windSpeed, 2, 5, 8);
   647→    else if (species === "карась") s = windSpeedScore(input.windSpeed, 1, 4, 7);
   648→    else s = windSpeedScore(input.windSpeed, 1, 5, 8);
   649→    results.push({ factor: "Скорость ветра", contribution: (w.windSpeed as number) * (s - 50) });
   650→  }
   651→  if (w.windDirection) {
   652→    const s = windDirScore(input.windDir, [180, 225, 270]);
   653→    results.push({ factor: "Направление ветра", contribution: (w.windDirection as number) * (s - 50) });
   654→  }
   655→  if (w.timeOfDay) {
   656→    let s: number;
   657→    if (species === "сом") {
   658→      s = input.isNight ? 95 : Math.max(20, timeOfDayScore(input.minutesFromSunEvent));
   659→    } else {
   660→      s = timeOfDayScore(input.minutesFromSunEvent);
   661→    }
   662→    results.push({ factor: "Время суток", contribution: (w.timeOfDay as number) * (s - 50) });
   663→  }
   664→  if (w.moonPhase) {
   665→    const s = moonPhaseScore(input.lunarDay);
   666→    results.push({ factor: "Фаза луны", contribution: (w.moonPhase as number) * (s - 50) });
   667→  }
   668→  if (w.cloudCover && species === "щука") {
   669→    const s = cloudCoverScore(input.cloudCover, "overcast");
   670→    results.push({ factor: "Облачность", contribution: (w.cloudCover as number) * (s - 50) });
   671→  }
   672→
   673→  results.sort((a, b) => Math.abs(b.contribution) - Math.abs(a.contribution));
   674→  return results.slice(0, 2).map((r) => r.factor);
   675→}

   679→// Mean air temperature over the past 3 days (first 72 hourly records, since
   680→// Open-Meteo returns past_days=3 prepended to forecast_days=7).
   681→export function meanPast3DaysAirTemp(temps: number[]): number {
   682→  if (temps.length === 0) return 15;
   683→  const past72 = temps.slice(0, 72);
   684→  const slice = past72.length > 0 ? past72 : temps;
   685→  return slice.reduce((s, v) => s + v, 0) / slice.length;
   686→}

   688→// Seasonal piecewise formula based on avg3 (mean of past 3 days air temp).
   689→// Rivers have lower thermal inertia (1-2 day lag) than lakes (5-7 day lag).
   690→export function estimateWaterTemp(avg3: number, waterType: "river" | "lake"): number {
   691→  let base: number;
   692→  if (avg3 < 4) {
   693→    // Winter: water stays above freezing
   694→    base = Math.max(0.5, 0.7 * avg3 + 2);
   695→  } else if (avg3 < 12) {
   696→    // Spring warming: water lags behind air
   697→    base = 0.8 * avg3 + 1;
   698→  } else if (avg3 < 20) {
   699→    // Transitional: water close to air
   700→    base = 0.95 * avg3;
   701→  } else {
   702→    // Summer: water slightly cooler than air (evaporative cooling)
   703→    base = 0.9 * avg3;
   704→  }
   705→  // Rivers flow → mix → more responsive (closer to air). Lakes retain heat → smooth.
   706→  if (waterType === "lake") {
   707→    base = base * 0.97;
   708→  }
   709→  return Math.max(0, Math.min(35, Math.round(base * 10) / 10));
   710→}
