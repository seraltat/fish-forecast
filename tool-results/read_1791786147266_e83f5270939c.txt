    17→// ----------------- Piecewise-linear interpolation -----------------
    18→
    19→// points sorted ascending by [0]; linear interp between; clamp ends.
    20→export function piecewiseLinear(points: Array<[number, number]>, x: number): number {
    21→  if (points.length === 0) return 0;
    22→  if (x <= points[0][0]) return points[0][1];
    23→  const last = points[points.length - 1];
    24→  if (x >= last[0]) return last[1];
    25→  for (let i = 0; i < points.length - 1; i++) {
    26→    const [x0, y0] = points[i];
    27→    const [x1, y1] = points[i + 1];
    28→    if (x >= x0 && x <= x1) {
    29→      const t = (x - x0) / (x1 - x0);
    30→      return Math.round(y0 + t * (y1 - y0));
    31→    }
    32→  }
    33→  return last[1];
    34→}

    42→export function moonPhase(date: Date): { age: number; phase: number; label: string } {
    43→  const ms = date.getTime() - LUNAR_EPOCH_MS;
    44→  const day = ms / (1000 * 60 * 60 * 24);
    45→  const age = ((day % SYNODIC_MONTH) + SYNODIC_MONTH) % SYNODIC_MONTH;
    46→  const phase = age / SYNODIC_MONTH; // 0..1
    47→  return { age, phase, label: moonPhaseLabel(phase) };
    48→}

   106→export type PressureTrend = "drop_sharp" | "drop" | "stable" | "rise" | "rise_sharp";

   108→// Thresholds based on user's research (image 1):
   109→//   Резкий скачок = падение ≥ 5–8 mmHg/сутки → ~0.6–1 mmHg/3ч
   110→//   Плавное падение = 2–3 mmHg/сутки → ~0.25–0.4 mmHg/3ч
   111→// Thresholds (per 3-hour window):
   112→//   < -1 mmHg/3ч           → Резко падает
   113→//   -1 .. -0.3 mmHg/3ч     → Падает (плавное)
   114→//   -0.3 .. +0.3 mmHg/3ч   → Стабильно
   115→//   +0.3 .. +1 mmHg/3ч     → Растёт (плавное)
   116→//   > +1 mmHg/3ч           → Резко растёт
   117→export function pressureTrend(deltaMmHg: number): { trend: PressureTrend; label: string } {
   118→  if (deltaMmHg < -1) return { trend: "drop_sharp", label: "Резко падает" };
   119→  if (deltaMmHg < -0.3) return { trend: "drop", label: "Падает" };
   120→  if (deltaMmHg <= 0.3) return { trend: "stable", label: "Стабильно" };
   121→  if (deltaMmHg <= 1) return { trend: "rise", label: "Растёт" };
   122→  return { trend: "rise_sharp", label: "Резко растёт" };
   123→}

   241→// Bell-shaped sub-score kept for backward compat (still used in some tests).
   242→export function bellScore(value: number, peak: number, lo: number, hi: number): number {
   243→  if (value < lo || value > hi) {
   244→    return Math.max(0, 100 - Math.abs(value - peak) * 8);
   245→  }
   246→  const half = Math.max((hi - lo) / 2, 0.001);
   247→  const d = Math.abs(value - peak) / half;
   248→  return Math.round(100 - 25 * d * d);
   249→}

   283→// Wind direction sub-score — piecewise by 8-point compass.
   284→// По правилу пользователя: «южный, западный клев нормальный; северный, восточный — не очень».
   285→// Используем 8-румбовый компас с явной оценкой для каждого направления:
   286→//   С  (N,   0°)   → 25  — холодный, не очень
   287→//   СВ (NE, 45°)  → 30  — холодный с востоком, не очень
   288→//   В  (E,  90°)  → 35  — восточный, не очень
   289→//   ЮВ (SE, 135°) → 75  — тёплый с востоком, хорошо
   290→//   Ю  (S, 180°)  → 90  — тёплый, хороший
   291→//   ЮЗ (SW, 225°) → 100 — лучший: тёплый, нагоняет корм в берег
   292→//   З  (W, 270°)  → 90  — хороший
   293→//   СЗ (NW, 315°) → 65  — западный с холодным севером, средне
   294→const WIND_DIR_SCORES_8PT: Array<{ short: string; score: number; deg: number }> = [
   295→  { short: "С", score: 25, deg: 0 },
   296→  { short: "СВ", score: 30, deg: 45 },
   297→  { short: "В", score: 35, deg: 90 },
   298→  { short: "ЮВ", score: 75, deg: 135 },
   299→  { short: "Ю", score: 90, deg: 180 },
   300→  { short: "ЮЗ", score: 100, deg: 225 },
   301→  { short: "З", score: 90, deg: 270 },
   302→  { short: "СЗ", score: 65, deg: 315 },
   303→];

   305→export function windDirScore(deg: number | null | undefined, _bestDirs?: number[]): number {
   306→  if (deg == null || Number.isNaN(deg)) return 50;
   307→  const norm = ((deg % 360) + 360) % 360;
   308→  // Find closest 8-point compass direction
   309→  let closest = WIND_DIR_SCORES_8PT[0];
   310→  let bestDiff = 360;
   311→  for (const dir of WIND_DIR_SCORES_8PT) {
   312→    const diff = Math.min(Math.abs(norm - dir.deg), 360 - Math.abs(norm - dir.deg));
   313→    if (diff < bestDiff) {
   314→      bestDiff = diff;
   315→      closest = dir;
   316→    }
   317→  }
   318→  // If exactly on the compass point, return its score.
   319→  // If between two points, interpolate linearly between their scores.
   320→  if (bestDiff === 0) return closest.score;
   321→  // Find the next compass point clockwise
   322→  const idx = WIND_DIR_SCORES_8PT.indexOf(closest);
   323→  const next = WIND_DIR_SCORES_8PT[(idx + 1) % 8];
   324→  const nextDeg = next.deg < closest.deg ? next.deg + 360 : next.deg;
   325→  const segmentSize = nextDeg - closest.deg;
   326→  const t = bestDiff / segmentSize;
   327→  return Math.round(closest.score + t * (next.score - closest.score));
   328→}

   363→// Convention: lunarDay = floor(moon age in days) — matches what's shown on popular sites
   364→// (Yandex Weather, Meteocast, Voshod-Solnca). Day 0 = day of new moon, day 1 = 1 day after, etc.
   365→//
   366→// Table (per user's research, shifted to Convention B):
   367→//   День 0-2 (новолуние ±): плохой → 25
   368→//   День 3: хороший → 70
   369→//   День 4-8 (первая четверть, растущая): отличный → 90
   370→//   День 9-10: хороший → 70
   371→//   День 11-12 (растущая, близко к полнолунию): слабый → 45
   372→//   День 13-16 (полнолуние ±): плохой → 25
   373→//   День 17 (сразу после полнолуния): слабый → 45
   374→//   День 18-19: отличный → 90
   375→//   День 20-22 (третья четверть): хороший → 70
   376→//   День 23-26 (убывающая): слабый → 45
   377→//   День 27+ (близко к новолунию): плохой → 25
   378→// Принимает ЛУННЫЙ ДЕНЬ (0..30) от computeUserLunarDay.
   379→export function moonPhaseScore(lunarDay: number): number {
   380→  const day = Math.max(0, Math.min(30, Math.floor(lunarDay)));
   381→  if (day <= 2) return 25;                       // плохой (новолуние ±)
   382→  if (day === 3) return 70;                     // хороший
   383→  if (day >= 4 && day <= 8) return 90;            // отличный
   384→  if (day === 9 || day === 10) return 70;         // хороший
   385→  if (day === 11 || day === 12) return 45;        // слабый
   386→  if (day >= 13 && day <= 16) return 25;          // плохой (полнолуние)
   387→  if (day === 17) return 45;                    // слабый
   388→  if (day === 18 || day === 19) return 90;      // отличный
   389→  if (day >= 20 && day <= 22) return 70;        // хороший
   390→  if (day >= 23 && day <= 26) return 45;        // слабый
   391→  if (day >= 27) return 25;                    // плохой (близко к новолунию)
   392→  return 50; // fallback
   393→}

   395→// Compute lunar day using the standard astronomical convention:
   396→//   lunarDay = floor(moon age in days)
   397→// where moon age = phase × SYNODIC_MONTH, and phase is the Open-Meteo value
   398→// (sampled at noon local for each day).
   399→//
   400→// This matches what's shown on popular sites (Yandex Weather, Meteocast, Voshod-Solnca):
   401→//   - On the day of new moon, age ≈ 0 → lunarDay = 0 (or "новолуние")
   402→//   - Day after new moon: age = 1.0+ → lunarDay = 1 ("1 день")
   403→//   - Day N after new moon: lunarDay = N
   404→//
   405→// No more moonrise-based counting — that was inconsistent (per-day phase variation
   406→// caused off-by-one errors when adjacent days' phases straddled an integer boundary).
   407→// Now the day number is stable: increments by 1 each solar day, never goes backwards.
   408→export function computeUserLunarDay(_now: Date, _moonrises: Date[], phase: number): number {
   409→  const age = phase * SYNODIC_MONTH;
   410→  return Math.floor(age);
   411→}
   412→
   413→// Find the most recent new moon before `now` using the lunar epoch + synodic month.
   414→// Iterates backwards from `now` to find the largest t such that
   415→// t = LUNAR_EPOCH_MS + k * SYNODIC_MONTH * ms_per_day, t <= now.
   416→export function mostRecentNewMoon(now: Date): Date {
   417→  const msPerDay = 1000 * 60 * 60 * 24;
   418→  const msPerCycle = SYNODIC_MONTH * msPerDay;
   419→  const elapsedMs = now.getTime() - LUNAR_EPOCH_MS;
   420→  const cycles = Math.floor(elapsedMs / msPerCycle);
   421→  return new Date(LUNAR_EPOCH_MS + cycles * msPerCycle);
   422→}

   424→// Wind speed sub-score: ideal [lo,hi], >badMps → low
   425→export function windSpeedScore(speed: number, lo: number, hi: number, badMps: number): number {
   426→  if (speed >= lo && speed <= hi) return 95;
   427→  if (speed > badMps) return 10;
   428→  if (speed < lo) {
   429→    const d = lo - speed;
   430→    return Math.max(20, Math.round(95 - d * 10));
   431→  }
   432→  const d = speed - hi;
   433→  return Math.max(10, Math.round(95 - d * 15));
   434→}

   436→// Cloud cover sub-score (0..100%)
   437→export function cloudCoverScore(cover: number, preference: "overcast" | "sunny" | "neutral"): number {
   438→  if (preference === "neutral") return 70;
   439→  if (preference === "overcast") {
   440→    return Math.round((cover / 100) * 90 + 10);
   441→  }
   442→  return Math.round(((100 - cover) / 100) * 90 + 10);
   443→}

   507→// Per-species scoring function
   508→export function scoreForSpecies(species: Species, weights: FishWeights, input: HourInput): number {
   509→  const subscores: Record<string, number> = {};
   510→
   511→  // Water temperature — piecewise curve per species.
   512→  if (weights.waterTemp) {
   513→    subscores.waterTemp = waterTempScore(species, input.waterTemp);
   514→  }
   515→
   516→  // Pressure trend — table-driven per species.
   517→  if (weights.pressureTrend) {
   518→    subscores.pressureTrend = pressureTrendScore(input.pressureTrend, species);
   519→  }
   520→
   521→  // Pressure value — same formula for all species (broad band, minor factor).
   522→  if (weights.pressureValue) {
   523→    subscores.pressureValue = pressureValueScore(input.pressureMmHg);
   524→  }
   525→
   526→  // Wind speed
   527→  if (weights.windSpeed) {
   528→    if (species === "сом") {
   529→      subscores.windSpeed = windSpeedScore(input.windSpeed, 1, 3, 5);
   530→    } else if (species === "щука") {
   531→      subscores.windSpeed = windSpeedScore(input.windSpeed, 2, 5, 8);
   532→    } else if (species === "карась") {
   533→      subscores.windSpeed = windSpeedScore(input.windSpeed, 1, 4, 7);
   534→    } else if (species === "сазан") {
   535→      subscores.windSpeed = windSpeedScore(input.windSpeed, 1, 5, 8);
   536→    }
   537→  }
   538→
   539→  // Wind direction
   540→  if (weights.windDirection) {
   541→    subscores.windDirection = windDirScore(input.windDir, [180, 225, 270]);
   542→  }
   543→
   544→  // Time of day
   545→  // Time of day — species-specific (was: pure Gaussian for all, which gave ~0 at noon
   546→  // even for щука — wrong, since predators hunt during the day).
   547→  // dawnDuskScore = Gaussian peak at sunrise/sunset (sigma=60 min).
   548→  // Базовые уровни (baseline) применяются, если Gaussian даёт меньше:
   549→  //   Щука днем: 65 (хищник, охотится весь день)
   550→  //   Карась/Сазан днем: 50 (клюёт, но слабее зорьки)
   551→  //   Сом днем: 25 (ночник, днём прячется)
   552→  // Ночью: щука=25 (визуальный охотник), карась/сазан=25, сом=95 (пик)
   553→  if (weights.timeOfDay) {
   554→    const dist = Math.min(input.minutesFromSunEvent, 720); // cap at 12h
   555→    const dawnDuskScore = 100 * Math.exp(-(dist * dist) / (60 * 60));
   556→
   557→    if (species === "сом") {
   558→      // Ночной хищник: пик ночью, переходные часы (рассвет/закат) — 80, днём — 25
   559→      subscores.timeOfDay = input.isNight
   560→        ? 95
   561→        : Math.max(25, Math.round(dawnDuskScore * 0.8));
   562→    } else if (species === "щука") {
   563→      // Хищник-засадчик: пик на рассвете/закате (90+), днём — 65 (продолжает охотиться),
   564→      // ночью — 25 (визуальный охотник, в темноте не видит)
   565→      subscores.timeOfDay = input.isNight
   566→        ? 25
   567→        : Math.round(Math.max(65, dawnDuskScore));
   568→    } else {
   569→      // Карась, сазан: пик на зорьке (90+), днём — 50 (клюёт, но слабее),
   570→      // ночью — 25 (неактивны в темноте)
   571→      subscores.timeOfDay = input.isNight
   572→        ? 25
   573→        : Math.round(Math.max(50, dawnDuskScore));
   574→    }
   575→  }
   576→
   577→  // Moon phase — одинаковая таблица 28-дневного цикла для всех видов рыб.
   578→  // Используем lunarDay (по восходам луны на координатах места) вместо
   579→  // глобальной фазы — это даёт корректный лунный день для русского календаря.
   580→  if (weights.moonPhase) {
   581→    subscores.moonPhase = moonPhaseScore(input.lunarDay);
   582→  }
   583→
   584→  // Cloud cover (only щука uses this)
   585→  if (weights.cloudCover) {
   586→    subscores.cloudCover = cloudCoverScore(input.cloudCover, "overcast");
   587→  }
   588→
   589→  // Weighted average
   590→  let total = 0;
   591→  let weightSum = 0;
   592→  for (const [key, w] of Object.entries(weights)) {
   593→    const wnum = w as number;
   594→    if (wnum && subscores[key] != null) {
   595→      total += wnum * subscores[key];
   596→      weightSum += wnum;
   597→    }
   598→  }
   599→  if (weightSum === 0) return 0;
   600→  let finalScore = Math.round(total / weightSum);
   601→
   602→  // Water-temperature gate — критически важная защита от абсурдных оценок.
   603→  // Если вода слишком холодная (или слишком тёплая) — рыба физически не активна,
   604→  // даже при идеальном давлении/ветре/времени суток. Множитель 0..1:
   605→  //   sub-score 50 (= порог активности) → gate 1.0 (без штрафа)
   606→  //   sub-score 25 (слабая активность)  → gate 0.5
   607→  //   sub-score 5  (рыба спит)         → gate 0.1 (сильный штраф)
   608→  // Это делает температуру воды «вратами» для клёва, а не просто одним из факторов.
   609→  if (subscores.waterTemp != null) {
   610→    const tempGate = Math.min(1, subscores.waterTemp / 50);
   611→    finalScore = Math.round(finalScore * tempGate);
   612→  }
   613→
   614→  return finalScore;
   615→}

   617→// Top contributing factors (for "Почему так?" tooltip).
   618→// Returns top 2 factors by weighted contribution * (subscore - 50).
   619→export function topFactors(species: Species, weights: FishWeights, input: HourInput): string[] {
   620→  const results: { factor: string; contribution: number }[] = [];
   621→
   622→  const w = weights;
   623→  if (w.waterTemp) {
   624→    const s = waterTempScore(species, input.waterTemp);
   625→    results.push({ factor: "Температура воды", contribution: (w.waterTemp as number) * (s - 50) });
   626→  }
   627→  if (w.pressureTrend) {
   628→    const s = pressureTrendScore(input.pressureTrend, species);
   629→    results.push({ factor: "Тенденция давления", contribution: (w.pressureTrend as number) * (s - 50) });
   630→  }
   631→  if (w.pressureValue) {
   632→    const s = pressureValueScore(input.pressureMmHg);
   633→    results.push({ factor: "Значение давления", contribution: (w.pressureValue as number) * (s - 50) });
   634→  }
   635→  if (w.windSpeed) {
   636→    let s: number;
   637→    if (species === "сом") s = windSpeedScore(input.windSpeed, 1, 3, 5);
   638→    else if (species === "щука") s = windSpeedScore(input.windSpeed, 2, 5, 8);
   639→    else if (species === "карась") s = windSpeedScore(input.windSpeed, 1, 4, 7);
   640→    else s = windSpeedScore(input.windSpeed, 1, 5, 8);
   641→    results.push({ factor: "Скорость ветра", contribution: (w.windSpeed as number) * (s - 50) });
   642→  }
   643→  if (w.windDirection) {
   644→    const s = windDirScore(input.windDir, [180, 225, 270]);
   645→    results.push({ factor: "Направление ветра", contribution: (w.windDirection as number) * (s - 50) });
   646→  }
   647→  if (w.timeOfDay) {
   648→    // Same species-specific logic as in scoreForSpecies (see comments there).
   649→    const dist = Math.min(input.minutesFromSunEvent, 720);
   650→    const dawnDuskScore = 100 * Math.exp(-(dist * dist) / (60 * 60));
   651→    let s: number;
   652→    if (species === "сом") {
   653→      s = input.isNight ? 95 : Math.max(25, Math.round(dawnDuskScore * 0.8));
   654→    } else if (species === "щука") {
   655→      s = input.isNight ? 25 : Math.round(Math.max(65, dawnDuskScore));
   656→    } else {
   657→      s = input.isNight ? 25 : Math.round(Math.max(50, dawnDuskScore));
   658→    }
   659→    results.push({ factor: "Время суток", contribution: (w.timeOfDay as number) * (s - 50) });
   660→  }
   661→  if (w.moonPhase) {
   662→    const s = moonPhaseScore(input.lunarDay);
   663→    results.push({ factor: "Фаза луны", contribution: (w.moonPhase as number) * (s - 50) });
   664→  }
   665→  if (w.cloudCover && species === "щука") {
   666→    const s = cloudCoverScore(input.cloudCover, "overcast");
   667→    results.push({ factor: "Облачность", contribution: (w.cloudCover as number) * (s - 50) });
   668→  }
   669→
   670→  results.sort((a, b) => Math.abs(b.contribution) - Math.abs(a.contribution));
   671→  return results.slice(0, 2).map((r) => r.factor);
   672→}

   674→// ----------------- Water temperature estimation -----------------

   676→// Mean air temperature over the past 3 days BEFORE the forecast starts.
   677→// Open-Meteo response: with past_days=35 + forecast_days=7, hourly array has 42*24=1008 hours.
   678→// Past 3 days before forecast = indices [N-168-72, N-168) = last 72 hours of past data.
   679→// If past_days=3 (legacy), the first 72 hours are the past 3 days — function still works.
   680→export function meanPast3DaysAirTemp(temps: number[]): number {
   681→  if (temps.length === 0) return 15;
   682→  // Last 72 hours BEFORE the last 168 (forecast) hours.
   683→  const forecastLen = 168;
   684→  let slice: number[];
   685→  if (temps.length > forecastLen + 72) {
   686→    // past_days >= 3: take the 72 hours right before forecast starts.
   687→    slice = temps.slice(temps.length - forecastLen - 72, temps.length - forecastLen);
   688→  } else if (temps.length >= 72) {
   689→    // Fallback: take the first 72 hours (assumes past_days=3, no forecast data prepended).
   690→    slice = temps.slice(0, 72);
   691→  } else {
   692→    slice = temps;
   693→  }
   694→  return slice.reduce((s, v) => s + v, 0) / slice.length;
   695→}

   697→// Seasonal piecewise formula based on avg3 (mean of past 3 days air temp).
   698→// Rivers have lower thermal inertia (1-2 day lag) than lakes (5-7 day lag).
   699→export function estimateWaterTemp(avg3: number, waterType: "river" | "lake"): number {
   700→  let base: number;
   701→  if (avg3 < 4) {
   702→    // Winter: water stays above freezing
   703→    base = Math.max(0.5, 0.7 * avg3 + 2);
   704→  } else if (avg3 < 12) {
   705→    // Spring warming: water lags behind air
   706→    base = 0.8 * avg3 + 1;
   707→  } else if (avg3 < 20) {
   708→    // Transitional: water close to air
   709→    base = 0.95 * avg3;
   710→  } else {
   711→    // Summer: water slightly cooler than air (evaporative cooling)
   712→    base = 0.9 * avg3;
   713→  }
   714→  // Rivers flow → mix → more responsive (closer to air). Lakes retain heat → smooth.
   715→  if (waterType === "lake") {
   716→    base = base * 0.97;
   717→  }
   718→  return Math.max(0, Math.min(35, Math.round(base * 10) / 10));
   719→}
