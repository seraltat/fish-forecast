    18→    18→
    19→    19→// points sorted ascending by [0]; linear interp between; clamp ends.
    20→    20→export function piecewiseLinear(points: Array<[number, number]>, x: number): number {
    21→    21→  if (points.length === 0) return 0;
    22→    22→  if (x <= points[0][0]) return points[0][1];
    23→    23→  const last = points[points.length - 1];
    24→    24→  if (x >= last[0]) return last[1];
    25→    25→  for (let i = 0; i < points.length - 1; i++) {
    26→    26→    const [x0, y0] = points[i];
    27→    27→    const [x1, y1] = points[i + 1];
    28→    28→    if (x >= x0 && x <= x1) {
    29→    29→      const t = (x - x0) / (x1 - x0);
    30→    30→      return Math.round(y0 + t * (y1 - y0));
    31→    31→    }
    32→    32→  }
    33→    33→  return last[1];
    34→    34→}

    42→    42→export function moonPhase(date: Date): { age: number; phase: number; label: string } {
    43→    43→  const ms = date.getTime() - LUNAR_EPOCH_MS;
    44→    44→  const day = ms / (1000 * 60 * 60 * 24);
    45→    45→  const age = ((day % SYNODIC_MONTH) + SYNODIC_MONTH) % SYNODIC_MONTH;
    46→    46→  const phase = age / SYNODIC_MONTH; // 0..1
    47→    47→  return { age, phase, label: moonPhaseLabel(phase) };
    48→    48→}

   107→   107→

   118→   118→  if (deltaMmHg < -1) return { trend: "drop_sharp", label: "Резко падает" };

   121→   121→  if (deltaMmHg <= 1) return { trend: "rise", label: "Растёт" };
   122→   122→  return { trend: "rise_sharp", label: "Резко растёт" };
   123→   123→}

   241→   241→// Bell-shaped sub-score kept for backward compat (still used in some tests).
   242→   242→export function bellScore(value: number, peak: number, lo: number, hi: number): number {
   243→   243→  if (value < lo || value > hi) {
   244→   244→    return Math.max(0, 100 - Math.abs(value - peak) * 8);
   245→   245→  }
   246→   246→  const half = Math.max((hi - lo) / 2, 0.001);
   247→   247→  const d = Math.abs(value - peak) / half;
   248→   248→  return Math.round(100 - 25 * d * d);
   249→   249→}

   283→   283→// Wind direction sub-score — piecewise by 8-point compass.
   284→   284→// По правилу пользователя: «южный, западный клев нормальный; северный, восточный — не очень».
   285→   285→// Используем 8-румбовый компас с явной оценкой для каждого направления:
   286→   286→//   С  (N,   0°)   → 25  — холодный, не очень
   287→   287→//   СВ (NE, 45°)  → 30  — холодный с востоком, не очень
   288→   288→//   В  (E,  90°)  → 35  — восточный, не очень
   289→   289→//   ЮВ (SE, 135°) → 75  — тёплый с востоком, хорошо
   290→   290→//   Ю  (S, 180°)  → 90  — тёплый, хороший
   291→   291→//   ЮЗ (SW, 225°) → 100 — лучший: тёплый, нагоняет корм в берег
   292→   292→//   З  (W, 270°)  → 90  — хороший
   293→   293→//   СЗ (NW, 315°) → 65  — западный с холодным севером, средне
   294→   294→const WIND_DIR_SCORES_8PT: Array<{ short: string; score: number; deg: number }> = [
   295→   295→  { short: "С", score: 25, deg: 0 },
   296→   296→  { short: "СВ", score: 30, deg: 45 },
   297→   297→  { short: "В", score: 35, deg: 90 },
   298→   298→  { short: "ЮВ", score: 75, deg: 135 },
   299→   299→  { short: "Ю", score: 90, deg: 180 },
   300→   300→  { short: "ЮЗ", score: 100, deg: 225 },
   301→   301→  { short: "З", score: 90, deg: 270 },
   302→   302→  { short: "СЗ", score: 65, deg: 315 },
   303→   303→];
   304→   304→
   305→   305→export function windDirScore(deg: number | null | undefined, _bestDirs?: number[]): number {
   306→   306→  if (deg == null || Number.isNaN(deg)) return 50;
   307→   307→  const norm = ((deg % 360) + 360) % 360;
   308→   308→  // Find closest 8-point compass direction
   309→   309→  let closest = WIND_DIR_SCORES_8PT[0];
   310→   310→  let bestDiff = 360;
   311→   311→  for (const dir of WIND_DIR_SCORES_8PT) {
   312→   312→    const diff = Math.min(Math.abs(norm - dir.deg), 360 - Math.abs(norm - dir.deg));
   313→   313→    if (diff < bestDiff) {
   314→   314→      bestDiff = diff;
   315→   315→      closest = dir;
   316→   316→    }
   317→   317→  }
   318→   318→  // If exactly on the compass point, return its score.
   319→   319→  // If between two points, interpolate linearly between their scores.
   320→   320→  if (bestDiff === 0) return closest.score;
   321→   321→  // Find the next compass point clockwise
   322→   322→  const idx = WIND_DIR_SCORES_8PT.indexOf(closest);
   323→   323→  const next = WIND_DIR_SCORES_8PT[(idx + 1) % 8];
   324→   324→  const nextDeg = next.deg < closest.deg ? next.deg + 360 : next.deg;
   325→   325→  const segmentSize = nextDeg - closest.deg;
   326→   326→  const t = bestDiff / segmentSize;
   327→   327→  return Math.round(closest.score + t * (next.score - closest.score));
   328→   328→}

   353→   353→// Lunar day number (1..30) from moon phase (0..1).
   354→   354→// Used for UI display — to show user which lunar day the algorithm is using.
   355→   355→// age = phase * 29.53 (days since new moon, 0-indexed).
   356→   356→// lunar day = floor(age) + 1 (1-indexed, day 1 = new moon).

   393→   393→// Compute Russian-fishing-tradition lunar day (1..30) for a given moment
   394→   394→// based on:
   395→   395→//   1. Astronomical new moon (computed from lunar epoch 2000-01-06 18:14 UTC)
   396→   396→//   2. List of moonrise times at the user's location (from Open-Meteo Astronomy API)
   397→   397→//
   398→   398→// Algorithm:
   399→   399→//   - Find the most recent new moon (date) before `now`
   400→   400→//   - Count moonrises that happened AFTER new moon AND BEFORE `now`
   401→   401→//   - lunarDay = count + 1
   402→   402→//   (Day 1 = short day from new moon moment to first moonrise)
   403→   403→//   (Day N = from (N-1)th to Nth moonrise after new moon)
   404→   404→//
   405→   405→// Parameters:
   406→   406→//   now — Date constructed from local time string treated as UTC (so getUTCHours() gives local hour)
   407→   407→//   moonrises — array of Date objects (any order). If empty, heuristic is used.
   408→   408→//   phase — moon phase (0..1) computed once per day at noon UTC (consistent across all hours of the day).
   409→   409→//            This is critical: recomputing phase from `now` would give a wrong value because
   410→   410→//            `now` represents local time (treated as UTC), not actual UTC moment.
   411→   411→export function computeUserLunarDay(now: Date, moonrises: Date[], phase: number): number {
   412→   412→  if (moonrises.length === 0) {
   413→   413→    // Heuristic fallback: estimate moonrise from phase.
   414→   414→    // At new moon (phase 0): moonrise ~ sunrise (~06:00 local).
   415→   415→    // At full moon (phase 0.5): moonrise ~ sunset (~18:00 local).
   416→   416→    // Linear interp: moonrise_hour = 6 + phase * 24 (mod 24).
   417→   417→    const age = phase * SYNODIC_MONTH;
   418→   418→    const astroDay = Math.floor(age) + 1;
   419→   419→    const moonriseHour = (6 + phase * 24) % 24; // local hour at spot
   420→   420→    // now.getUTCHours() gives the local hour at the spot because we constructed
   421→   421→    // the Date from Open-Meteo local time strings treated as UTC.
   422→   422→    const localHour = now.getUTCHours() + now.getUTCMinutes() / 60;
   423→   423→    // Account for moonrise that may be early morning (e.g. 02:00) and now is 22:00
   424→   424→    // — that means we're past moonrise, so add 1.
   425→   425→    if (localHour >= moonriseHour) {
   426→   426→      return astroDay + 1;
   427→   427→    }
   428→   428→    return astroDay;
   429→   429→  }
   430→   430→
   431→   431→  // Find most recent new moon before `now`.
   432→   432→  const newMoonDate = mostRecentNewMoon(now);
   433→   433→  // Count moonrises strictly AFTER new moon AND BEFORE (or at) now.
   434→   434→  const count = moonrises.filter(
   435→   435→    (mr) => mr.getTime() > newMoonDate.getTime() && mr.getTime() <= now.getTime()
   436→   436→  ).length;
   437→   437→
   438→   438→  return count + 1;
   439→   439→}
   440→   440→
   441→   441→// Find the most recent new moon before `now` using the lunar epoch + synodic month.
   442→   442→// Iterates backwards from `now` to find the largest t such that
   443→   443→// t = LUNAR_EPOCH_MS + k * SYNODIC_MONTH * ms_per_day, t <= now.
   444→   444→export function mostRecentNewMoon(now: Date): Date {
   445→   445→  const msPerDay = 1000 * 60 * 60 * 24;
   446→   446→  const msPerCycle = SYNODIC_MONTH * msPerDay;
   447→   447→  const elapsedMs = now.getTime() - LUNAR_EPOCH_MS;
   448→   448→  const cycles = Math.floor(elapsedMs / msPerCycle);
   449→   449→  return new Date(LUNAR_EPOCH_MS + cycles * msPerCycle);
   450→   450→}

   452→   452→// Wind speed sub-score: ideal [lo,hi], >badMps → low
   453→   453→export function windSpeedScore(speed: number, lo: number, hi: number, badMps: number): number {
   454→   454→  if (speed >= lo && speed <= hi) return 95;
   455→   455→  if (speed > badMps) return 10;
   456→   456→  if (speed < lo) {
   457→   457→    const d = lo - speed;
   458→   458→    return Math.max(20, Math.round(95 - d * 10));
   459→   459→  }
   460→   460→  const d = speed - hi;
   461→   461→  return Math.max(10, Math.round(95 - d * 15));
   462→   462→}

   536→   536→export function scoreForSpecies(species: Species, weights: FishWeights, input: HourInput): number {
   537→   537→  const subscores: Record<string, number> = {};
   538→   538→
   539→   539→  // Water temperature — piecewise curve per species.
   540→   540→  if (weights.waterTemp) {
   541→   541→    subscores.waterTemp = waterTempScore(species, input.waterTemp);
   542→   542→  }
   543→   543→
   544→   544→  // Pressure trend — table-driven per species.
   545→   545→  if (weights.pressureTrend) {
   546→   546→    subscores.pressureTrend = pressureTrendScore(input.pressureTrend, species);
   547→   547→  }
   548→   548→
   549→   549→  // Pressure value — same formula for all species (broad band, minor factor).
   550→   550→  if (weights.pressureValue) {
   551→   551→    subscores.pressureValue = pressureValueScore(input.pressureMmHg);
   552→   552→  }
   553→   553→
   554→   554→  // Wind speed
   555→   555→  if (weights.windSpeed) {
   556→   556→    if (species === "сом") {
   557→   557→      subscores.windSpeed = windSpeedScore(input.windSpeed, 1, 3, 5);
   558→   558→    } else if (species === "щука") {

   560→   560→    } else if (species === "карась") {
   561→   561→      subscores.windSpeed = windSpeedScore(input.windSpeed, 1, 4, 7);
   562→   562→    } else if (species === "сазан") {
   563→   563→      subscores.windSpeed = windSpeedScore(input.windSpeed, 1, 5, 8);
   564→   564→    }
   565→   565→  }
   566→   566→
   567→   567→  // Wind direction
   568→   568→  if (weights.windDirection) {
   569→   569→    subscores.windDirection = windDirScore(input.windDir, [180, 225, 270]);
   570→   570→  }
   571→   571→
   572→   572→  // Time of day
   573→   573→  if (weights.timeOfDay) {
   574→   574→    if (species === "сом") {
   575→   575→      if (input.isNight) {
   576→   576→        subscores.timeOfDay = 95;
   577→   577→      } else {
   578→   578→        subscores.timeOfDay = Math.max(20, timeOfDayScore(input.minutesFromSunEvent));
   579→   579→      }
   580→   580→    } else {
   581→   581→      subscores.timeOfDay = timeOfDayScore(input.minutesFromSunEvent);
   582→   582→    }
   583→   583→  }
   584→   584→
   585→   585→  // Moon phase — одинаковая таблица 28-дневного цикла для всех видов рыб.
   586→   586→  // Используем lunarDay (по восходам луны на координатах места) вместо
   587→   587→  // глобальной фазы — это даёт корректный лунный день для русского календаря.
   588→   588→  if (weights.moonPhase) {
   589→   589→    subscores.moonPhase = moonPhaseScore(input.lunarDay);
   590→   590→  }
   591→   591→
   592→   592→  // Cloud cover (only щука uses this)
   593→   593→  if (weights.cloudCover) {
   594→   594→    subscores.cloudCover = cloudCoverScore(input.cloudCover, "overcast");
   595→   595→  }
   596→   596→
   597→   597→  // Weighted average
   598→   598→  let total = 0;
   599→   599→  let weightSum = 0;
   600→   600→  for (const [key, w] of Object.entries(weights)) {
   601→   601→    const wnum = w as number;
   602→   602→    if (wnum && subscores[key] != null) {
   603→   603→      total += wnum * subscores[key];
   604→   604→      weightSum += wnum;
   605→   605→    }
   606→   606→  }
   607→   607→  if (weightSum === 0) return 0;
   608→   608→  let finalScore = Math.round(total / weightSum);
   609→   609→
   610→   610→  // Water-temperature gate — критически важная защита от абсурдных оценок.
   611→   611→  // Если вода слишком холодная (или слишком тёплая) — рыба физически не активна,
   612→   612→  // даже при идеальном давлении/ветре/времени суток. Множитель 0..1:
   613→   613→  //   sub-score 50 (= порог активности) → gate 1.0 (без штрафа)
   614→   614→  //   sub-score 25 (слабая активность)  → gate 0.5
   615→   615→  //   sub-score 5  (рыба спит)         → gate 0.1 (сильный штраф)
   616→   616→  // Это делает температуру воды «вратами» для клёва, а не просто одним из факторов.
   617→   617→  if (subscores.waterTemp != null) {
   618→   618→    const tempGate = Math.min(1, subscores.waterTemp / 50);
   619→   619→    finalScore = Math.round(finalScore * tempGate);
   620→   620→  }
   621→   621→
   622→   622→  return finalScore;
   623→   623→}

   683→   683→  const past72 = temps.slice(0, 72);

   688→   688→// Seasonal piecewise formula based on avg3 (mean of past 3 days air temp).
   689→   689→// Rivers have lower thermal inertia (1-2 day lag) than lakes (5-7 day lag).
   690→   690→export function estimateWaterTemp(avg3: number, waterType: "river" | "lake"): number {
   691→   691→  let base: number;
   692→   692→  if (avg3 < 4) {
   693→   693→    // Winter: water stays above freezing
   694→   694→    base = Math.max(0.5, 0.7 * avg3 + 2);
   695→   695→  } else if (avg3 < 12) {
   696→   696→    // Spring warming: water lags behind air
   697→   697→    base = 0.8 * avg3 + 1;
   698→   698→  } else if (avg3 < 20) {
   699→   699→    // Transitional: water close to air
   700→   700→    base = 0.95 * avg3;
   701→   701→  } else {
   702→   702→    // Summer: water slightly cooler than air (evaporative cooling)
   703→   703→    base = 0.9 * avg3;
   704→   704→  }
   705→   705→  // Rivers flow → mix → more responsive (closer to air). Lakes retain heat → smooth.
   706→   706→  if (waterType === "lake") {
   707→   707→    base = base * 0.97;
   708→   708→  }
   709→   709→  return Math.max(0, Math.min(35, Math.round(base * 10) / 10));
   710→   710→}
