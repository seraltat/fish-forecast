    17→// ----------------- Piecewise-linear interpolation -----------------
    18→
    19→// points sorted ascending by [0]; linear interp between; clamp ends.
    20→export function piecewiseLinear(points: Array<[number, number]>, x: number): number {
    21→  if (points.length === 0) return 0;
    22→  if (x <= points[0][0]) return points[0][1];
    23→  const last = points[points.length - 1];
    24→  if (x >= last[0]) return last[1];
    25→  for (let i = 0; i < points.length - 1; i++) {
    26→    const [x0, y0] = points[i];
    27→    const [x1, y1] = points[i + 1];
    28→    if (x >= x0 && x <= x1) {
    29→      const t = (x - x0) / (x1 - x0);
    30→      return Math.round(y0 + t * (y1 - y0));
    31→    }
    32→  }
    33→  return last[1];
    34→}

    42→export function moonPhase(date: Date): { age: number; phase: number; label: string } {
    43→  const ms = date.getTime() - LUNAR_EPOCH_MS;
    44→  const day = ms / (1000 * 60 * 60 * 24);
    45→  const age = ((day % SYNODIC_MONTH) + SYNODIC_MONTH) % SYNODIC_MONTH;
    46→  const phase = age / SYNODIC_MONTH; // 0..1
    47→  return { age, phase, label: moonPhaseLabel(phase) };
    48→}

   106→export type PressureTrend = "drop_sharp" | "drop" | "stable" | "rise" | "rise_sharp";
   107→
   108→// Thresholds based on user's research (image 1):
   109→//   Резкий скачок = падение ≥ 5–8 mmHg/сутки → ~0.6–1 mmHg/3ч
   110→//   Плавное падение = 2–3 mmHg/сутки → ~0.25–0.4 mmHg/3ч
   111→// Thresholds (per 3-hour window):
   112→//   < -1 mmHg/3ч           → Резко падает
   113→//   -1 .. -0.3 mmHg/3ч     → Падает (плавное)
   114→//   -0.3 .. +0.3 mmHg/3ч   → Стабильно
   115→//   +0.3 .. +1 mmHg/3ч     → Растёт (плавное)
   116→//   > +1 mmHg/3ч           → Резко растёт
   117→export function pressureTrend(deltaMmHg: number): { trend: PressureTrend; label: string } {
   118→  if (deltaMmHg < -1) return { trend: "drop_sharp", label: "Резко падает" };
   119→  if (deltaMmHg < -0.3) return { trend: "drop", label: "Падает" };
   120→  if (deltaMmHg <= 0.3) return { trend: "stable", label: "Стабильно" };
   121→  if (deltaMmHg <= 1) return { trend: "rise", label: "Растёт" };
   122→  return { trend: "rise_sharp", label: "Резко растёт" };
   123→}

   242→// Bell-shaped sub-score kept for backward compat (still used in some tests).
   243→export function bellScore(value: number, peak: number, lo: number, hi: number): number {
   244→  if (value < lo || value > hi) {
   245→    return Math.max(0, 100 - Math.abs(value - peak) * 8);
   246→  }
   247→  const half = Math.max((hi - lo) / 2, 0.001);
   248→  const d = Math.abs(value - peak) / half;
   249→  return Math.round(100 - 25 * d * d);
   250→}

   278→export function timeOfDayScore(minutesFromSunEvent: number): number {
   279→  const sigma = 60;
   280→  const v = 100 * Math.exp(-(minutesFromSunEvent * minutesFromSunEvent) / (sigma * sigma));

   284→// Wind direction sub-score — piecewise by 8-point compass.
   285→// По правилу пользователя: «южный, западный клев нормальный; северный, восточный — не очень».
   286→// Используем 8-румбовый компас с явной оценкой для каждого направления:
   287→//   С  (N,   0°)   → 25  — холодный, не очень
   288→//   СВ (NE, 45°)  → 30  — холодный с востоком, не очень
   289→//   В  (E,  90°)  → 35  — восточный, не очень
   290→//   ЮВ (SE, 135°) → 75  — тёплый с востоком, хорошо
   291→//   Ю  (S, 180°)  → 90  — тёплый, хороший
   292→//   ЮЗ (SW, 225°) → 100 — лучший: тёплый, нагоняет корм в берег
   293→//   З  (W, 270°)  → 90  — хороший
   294→//   СЗ (NW, 315°) → 65  — западный с холодным севером, средне
   295→const WIND_DIR_SCORES_8PT: Array<{ short: string; score: number; deg: number }> = [
   296→  { short: "С", score: 25, deg: 0 },
   297→  { short: "СВ", score: 30, deg: 45 },
   298→  { short: "В", score: 35, deg: 90 },
   299→  { short: "ЮВ", score: 75, deg: 135 },
   300→  { short: "Ю", score: 90, deg: 180 },
   301→  { short: "ЮЗ", score: 100, deg: 225 },
   302→  { short: "З", score: 90, deg: 270 },
   303→  { short: "СЗ", score: 65, deg: 315 },
   304→];
   305→
   306→export function windDirScore(deg: number | null | undefined, _bestDirs?: number[]): number {
   307→  if (deg == null || Number.isNaN(deg)) return 50;
   308→  const norm = ((deg % 360) + 360) % 360;
   309→  // Find closest 8-point compass direction
   310→  let closest = WIND_DIR_SCORES_8PT[0];
   311→  let bestDiff = 360;
   312→  for (const dir of WIND_DIR_SCORES_8PT) {
   313→    const diff = Math.min(Math.abs(norm - dir.deg), 360 - Math.abs(norm - dir.deg));
   314→    if (diff < bestDiff) {
   315→      bestDiff = diff;
   316→      closest = dir;
   317→    }
   318→  }
   319→  // If exactly on the compass point, return its score.
   320→  // If between two points, interpolate linearly between their scores.
   321→  if (bestDiff === 0) return closest.score;
   322→  // Find the next compass point clockwise
   323→  const idx = WIND_DIR_SCORES_8PT.indexOf(closest);
   324→  const next = WIND_DIR_SCORES_8PT[(idx + 1) % 8];
   325→  const nextDeg = next.deg < closest.deg ? next.deg + 360 : next.deg;
   326→  const segmentSize = nextDeg - closest.deg;
   327→  const t = bestDiff / segmentSize;
   328→  return Math.round(closest.score + t * (next.score - closest.score));
   329→}

   384→  if (isOffSeason && waterTemp < 8) {

   391→    }

   393→      return {
   394→        status: "off_season",
   395→        label: "Зимовка",
   396→        description: "Сом залёг в зимовальные ямы и не питается. Сезон откроется в мае при прогреве воды до 8-10°C"
   397→      };
   398→    }

   402→      label: "Не сезон",

   405→  }
   406→

   609→// Lunar day number (1..30) from moon phase (0..1).
   610→// Used for UI display — to show user which lunar day the algorithm is using.
   611→// age = phase * 29.53 (days since new moon, 0-indexed).
   612→// lunar day = floor(age) + 1 (1-indexed, day 1 = new moon).

   618→// Moon phase sub-score — based on user's research (28-day lunar cycle table).
   619→// Convention: lunarDay = floor(moon age in days) — matches what's shown on popular sites
   620→// (Yandex Weather, Meteocast, Voshod-Solnca). Day 0 = day of new moon, day 1 = 1 day after, etc.
   621→//
   622→// Table (per user's research, shifted to Convention B):
   623→//   День 0-2 (новолуние ±): плохой → 25
   624→//   День 3: хороший → 70
   625→//   День 4-8 (первая четверть, растущая): отличный → 90
   626→//   День 9-10: хороший → 70
   627→//   День 11-12 (растущая, близко к полнолунию): слабый → 45
   628→//   День 13-16 (полнолуние ±): плохой → 25
   629→//   День 17 (сразу после полнолуния): слабый → 45
   630→//   День 18-19: отличный → 90
   631→//   День 20-22 (третья четверть): хороший → 70
   632→//   День 23-26 (убывающая): слабый → 45
   633→//   День 27+ (близко к новолунию): плохой → 25
   634→// Принимает ЛУННЫЙ ДЕНЬ (0..30) от computeUserLunarDay.
   635→export function moonPhaseScore(lunarDay: number): number {
   636→  const day = Math.max(0, Math.min(30, Math.floor(lunarDay)));
   637→  if (day <= 2) return 25;                       // плохой (новолуние ±)
   638→  if (day === 3) return 70;                     // хороший
   639→  if (day >= 4 && day <= 8) return 90;            // отличный
   640→  if (day === 9 || day === 10) return 70;         // хороший
   641→  if (day === 11 || day === 12) return 45;        // слабый
   642→  if (day >= 13 && day <= 16) return 25;          // плохой (полнолуние)
   643→  if (day === 17) return 45;                    // слабый
   644→  if (day === 18 || day === 19) return 90;      // отличный
   645→  if (day >= 20 && day <= 22) return 70;        // хороший
   646→  if (day >= 23 && day <= 26) return 45;        // слабый
   647→  if (day >= 27) return 25;                    // плохой (близко к новолунию)
   648→  return 50; // fallback
   649→}

   651→// Compute lunar day using the standard astronomical convention:
   652→//   lunarDay = floor(moon age in days)
   653→// where moon age = phase × SYNODIC_MONTH, and phase is the Open-Meteo value
   654→// (sampled at noon local for each day).
   655→//
   656→// This matches what's shown on popular sites (Yandex Weather, Meteocast, Voshod-Solnca):
   657→//   - On the day of new moon, age ≈ 0 → lunarDay = 0 (or "новолуние")
   658→//   - Day after new moon: age = 1.0+ → lunarDay = 1 ("1 день")
   659→//   - Day N after new moon: lunarDay = N
   660→//
   661→// No more moonrise-based counting — that was inconsistent (per-day phase variation
   662→// caused off-by-one errors when adjacent days' phases straddled an integer boundary).
   663→// Now the day number is stable: increments by 1 each solar day, never goes backwards.
   664→export function computeUserLunarDay(_now: Date, _moonrises: Date[], phase: number): number {
   665→  const age = phase * SYNODIC_MONTH;
   666→  return Math.floor(age);
   667→}

   669→// Find the most recent new moon before `now` using the lunar epoch + synodic month.
   670→// Iterates backwards from `now` to find the largest t such that
   671→// t = LUNAR_EPOCH_MS + k * SYNODIC_MONTH * ms_per_day, t <= now.
   672→export function mostRecentNewMoon(now: Date): Date {
   673→  const msPerDay = 1000 * 60 * 60 * 24;
   674→  const msPerCycle = SYNODIC_MONTH * msPerDay;
   675→  const elapsedMs = now.getTime() - LUNAR_EPOCH_MS;
   676→  const cycles = Math.floor(elapsedMs / msPerCycle);
   677→  return new Date(LUNAR_EPOCH_MS + cycles * msPerCycle);
   678→}

   680→// Wind speed sub-score: ideal [lo,hi], >badMps → low
   681→export function windSpeedScore(speed: number, lo: number, hi: number, badMps: number): number {
   682→  if (speed >= lo && speed <= hi) return 95;
   683→  if (speed > badMps) return 10;
   684→  if (speed < lo) {
   685→    const d = lo - speed;
   686→    return Math.max(20, Math.round(95 - d * 10));
   687→  }
   688→  const d = speed - hi;
   689→  return Math.max(10, Math.round(95 - d * 15));
   690→}

   692→// Cloud cover sub-score (0..100%)
   693→export function cloudCoverScore(cover: number, preference: "overcast" | "sunny" | "neutral"): number {
   694→  if (preference === "neutral") return 70;
   695→  if (preference === "overcast") {
   696→    return Math.round((cover / 100) * 90 + 10);
   697→  }
   698→  return Math.round(((100 - cover) / 100) * 90 + 10);
   699→}

   764→// Per-species scoring function
   765→export function scoreForSpecies(species: Species, weights: FishWeights, input: HourInput): number {
   766→  const subscores: Record<string, number> = {};
   767→
   768→  // Water temperature — piecewise curve per species.
   769→  if (weights.waterTemp) {
   770→    subscores.waterTemp = waterTempScore(species, input.waterTemp);
   771→  }
   772→
   773→  // Pressure trend — table-driven per species.
   774→  if (weights.pressureTrend) {
   775→    subscores.pressureTrend = pressureTrendScore(input.pressureTrend, species);
   776→  }
   777→
   778→  // Pressure value — same formula for all species (broad band, minor factor).
   779→  if (weights.pressureValue) {
   780→    subscores.pressureValue = pressureValueScore(input.pressureMmHg);
   781→  }
   782→
   783→  // Wind speed
   784→  if (weights.windSpeed) {
   785→    if (species === "сом") {
   786→      subscores.windSpeed = windSpeedScore(input.windSpeed, 1, 3, 5);
   787→    } else if (species === "щука") {
   788→      subscores.windSpeed = windSpeedScore(input.windSpeed, 2, 5, 8);
   789→    } else if (species === "карась") {
   790→      subscores.windSpeed = windSpeedScore(input.windSpeed, 1, 4, 7);
   791→    } else if (species === "сазан") {
   792→      subscores.windSpeed = windSpeedScore(input.windSpeed, 1, 5, 8);
   793→    }
   794→  }
   795→
   796→  // Wind direction
   797→  if (weights.windDirection) {
   798→    subscores.windDirection = windDirScore(input.windDir, [180, 225, 270]);
   799→  }
   800→
   801→  // Time of day
   802→  // Time of day — species-specific (was: pure Gaussian for all, which gave ~0 at noon
   803→  // even for щука — wrong, since predators hunt during the day).
   804→  // dawnDuskScore = Gaussian peak at sunrise/sunset (sigma=60 min).
   805→  // Базовые уровни (baseline) применяются, если Gaussian даёт меньше:
   806→  //   Щука днем: 65 (хищник, охотится весь день)
   807→  //   Карась/Сазан днем: 50 (клюёт, но слабее зорьки)
   808→  //   Сом днем: 25 (ночник, днём прячется)
   809→  // Ночью: щука=25 (визуальный охотник), карась/сазан=25, сом=95 (пик)
   810→  if (weights.timeOfDay) {
   811→    const dist = Math.min(input.minutesFromSunEvent, 720); // cap at 12h
   812→    const dawnDuskScore = 100 * Math.exp(-(dist * dist) / (60 * 60));
   813→
   814→    if (species === "сом") {
   815→      // Ночной хищник: пик ночью, переходные часы (рассвет/закат) — 80, днём — 25
   816→      subscores.timeOfDay = input.isNight
   817→        ? 95
   818→        : Math.max(25, Math.round(dawnDuskScore * 0.8));
   819→    } else if (species === "щука") {
   820→      // Хищник-засадчик: пик на рассвете/закате (90+), днём — 65 (продолжает охотиться),
   821→      // ночью — 25 (визуальный охотник, в темноте не видит)
   822→      subscores.timeOfDay = input.isNight
   823→        ? 25
   824→        : Math.round(Math.max(65, dawnDuskScore));
   825→    } else {
   826→      // Карась, сазан: пик на зорьке (90+), днём — 50 (клюёт, но слабее),
   827→      // ночью — 25 (неактивны в темноте)
   828→      subscores.timeOfDay = input.isNight
   829→        ? 25
   830→        : Math.round(Math.max(50, dawnDuskScore));
   831→    }
   832→  }
   833→
   834→  // Moon phase — одинаковая таблица 28-дневного цикла для всех видов рыб.
   835→  // Используем lunarDay (по восходам луны на координатах места) вместо
   836→  // глобальной фазы — это даёт корректный лунный день для русского календаря.
   837→  if (weights.moonPhase) {
   838→    subscores.moonPhase = moonPhaseScore(input.lunarDay);
   839→  }
   840→
   841→  // Cloud cover (only щука uses this)
   842→  if (weights.cloudCover) {
   843→    subscores.cloudCover = cloudCoverScore(input.cloudCover, "overcast");
   844→  }
   845→
   846→  // Weighted average
   847→  let total = 0;
   848→  let weightSum = 0;
   849→  for (const [key, w] of Object.entries(weights)) {
   850→    const wnum = w as number;
   851→    if (wnum && subscores[key] != null) {
   852→      total += wnum * subscores[key];
   853→      weightSum += wnum;
   854→    }
   855→  }
   856→  if (weightSum === 0) return 0;
   857→  let finalScore = Math.round(total / weightSum);
   858→
   859→  // Water-temperature gate — критически важная защита от абсурдных оценок.
   860→  // Если вода слишком холодная (или слишком тёплая) — рыба физически не активна,
   861→  // даже при идеальном давлении/ветре/времени суток. Множитель 0..1:
   862→  //   sub-score 50 (= порог активности) → gate 1.0 (без штрафа)
   863→  //   sub-score 25 (слабая активность)  → gate 0.5
   864→  //   sub-score 5  (рыба спит)         → gate 0.1 (сильный штраф)
   865→  // Это делает температуру воды «вратами» для клёва, а не просто одним из факторов.
   866→  if (subscores.waterTemp != null) {
   867→    const tempGate = Math.min(1, subscores.waterTemp / 50);
   868→    finalScore = Math.round(finalScore * tempGate);
   869→  }
   870→
   871→  // Apply season modifier (for карась and сазан — off-season, spawning, etc.)
   872→  // сом and щука always return 1.0 (no seasonal rules).
   873→  const season = getSeasonStatus(species, input.month, input.waterTemp);
   874→  finalScore = Math.round(finalScore * seasonModifier(season.status));
   875→
   876→  return Math.min(100, Math.max(0, finalScore));
   877→}

   879→// Top contributing factors (for "Почему так?" tooltip).
   880→// Returns top 2 factors by weighted contribution * (subscore - 50).
   881→export function topFactors(species: Species, weights: FishWeights, input: HourInput): string[] {
   882→  const results: { factor: string; contribution: number }[] = [];
   883→
   884→  const w = weights;
   885→  if (w.waterTemp) {
   886→    const s = waterTempScore(species, input.waterTemp);
   887→    results.push({ factor: "Температура воды", contribution: (w.waterTemp as number) * (s - 50) });
   888→  }
   889→  if (w.pressureTrend) {
   890→    const s = pressureTrendScore(input.pressureTrend, species);
   891→    results.push({ factor: "Тенденция давления", contribution: (w.pressureTrend as number) * (s - 50) });
   892→  }
   893→  if (w.pressureValue) {
   894→    const s = pressureValueScore(input.pressureMmHg);
   895→    results.push({ factor: "Значение давления", contribution: (w.pressureValue as number) * (s - 50) });
   896→  }
   897→  if (w.windSpeed) {
   898→    let s: number;
   899→    if (species === "сом") s = windSpeedScore(input.windSpeed, 1, 3, 5);
   900→    else if (species === "щука") s = windSpeedScore(input.windSpeed, 2, 5, 8);
   901→    else if (species === "карась") s = windSpeedScore(input.windSpeed, 1, 4, 7);
   902→    else s = windSpeedScore(input.windSpeed, 1, 5, 8);
   903→    results.push({ factor: "Скорость ветра", contribution: (w.windSpeed as number) * (s - 50) });
   904→  }
   905→  if (w.windDirection) {
   906→    const s = windDirScore(input.windDir, [180, 225, 270]);
   907→    results.push({ factor: "Направление ветра", contribution: (w.windDirection as number) * (s - 50) });
   908→  }
   909→  if (w.timeOfDay) {
   910→    // Same species-specific logic as in scoreForSpecies (see comments there).
   911→    const dist = Math.min(input.minutesFromSunEvent, 720);
   912→    const dawnDuskScore = 100 * Math.exp(-(dist * dist) / (60 * 60));
   913→    let s: number;
   914→    if (species === "сом") {
   915→      s = input.isNight ? 95 : Math.max(25, Math.round(dawnDuskScore * 0.8));
   916→    } else if (species === "щука") {
   917→      s = input.isNight ? 25 : Math.round(Math.max(65, dawnDuskScore));
   918→    } else {
   919→      s = input.isNight ? 25 : Math.round(Math.max(50, dawnDuskScore));
   920→    }
   921→    results.push({ factor: "Время суток", contribution: (w.timeOfDay as number) * (s - 50) });
   922→  }
   923→  if (w.moonPhase) {
   924→    const s = moonPhaseScore(input.lunarDay);
   925→    results.push({ factor: "Фаза луны", contribution: (w.moonPhase as number) * (s - 50) });
   926→  }
   927→  if (w.cloudCover && species === "щука") {
   928→    const s = cloudCoverScore(input.cloudCover, "overcast");
   929→    results.push({ factor: "Облачность", contribution: (w.cloudCover as number) * (s - 50) });
   930→  }
   931→
   932→  results.sort((a, b) => Math.abs(b.contribution) - Math.abs(a.contribution));
   933→  return results.slice(0, 2).map((r) => r.factor);
   934→}

   938→// Mean air temperature over the past 3 days BEFORE the forecast starts.
   939→// Open-Meteo response: with past_days=35 + forecast_days=7, hourly array has 42*24=1008 hours.
   940→// Past 3 days before forecast = indices [N-168-72, N-168) = last 72 hours of past data.
   941→// If past_days=3 (legacy), the first 72 hours are the past 3 days — function still works.
   942→export function meanPast3DaysAirTemp(temps: number[]): number {
   943→  if (temps.length === 0) return 15;
   944→  // Last 72 hours BEFORE the last 168 (forecast) hours.
   945→  const forecastLen = 168;
   946→  let slice: number[];
   947→  if (temps.length > forecastLen + 72) {
   948→    // past_days >= 3: take the 72 hours right before forecast starts.
   949→    slice = temps.slice(temps.length - forecastLen - 72, temps.length - forecastLen);
   950→  } else if (temps.length >= 72) {
   951→    // Fallback: take the first 72 hours (assumes past_days=3, no forecast data prepended).
   952→    slice = temps.slice(0, 72);
   953→  } else {
   954→    slice = temps;
   955→  }
   956→  return slice.reduce((s, v) => s + v, 0) / slice.length;
   957→}

   959→// Seasonal piecewise formula based on avg3 (mean of past 3 days air temp).
   960→// Rivers have lower thermal inertia (1-2 day lag) than lakes (5-7 day lag).
   961→export function estimateWaterTemp(avg3: number, waterType: "river" | "lake"): number {
   962→  let base: number;
   963→  if (avg3 < 4) {
   964→    // Winter: water stays above freezing
   965→    base = Math.max(0.5, 0.7 * avg3 + 2);
   966→  } else if (avg3 < 12) {
   967→    // Spring warming: water lags behind air
   968→    base = 0.8 * avg3 + 1;
   969→  } else if (avg3 < 20) {
   970→    // Transitional: water close to air
   971→    base = 0.95 * avg3;
   972→  } else {
   973→    // Summer: water slightly cooler than air (evaporative cooling)
   974→    base = 0.9 * avg3;
   975→  }
   976→  // Rivers flow → mix → more responsive (closer to air). Lakes retain heat → smooth.
   977→  if (waterType === "lake") {
   978→    base = base * 0.97;
   979→  }
   980→  return Math.max(0, Math.min(35, Math.round(base * 10) / 10));
   981→}
